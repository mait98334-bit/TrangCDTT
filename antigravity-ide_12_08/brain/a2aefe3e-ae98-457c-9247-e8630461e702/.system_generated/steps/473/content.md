    replaceCartItemThumbnail: false,
    showFontNameInCart: true,
    showFontColorInCart: true,
    displayLayerNameInCart: true,
    autofixerDelay: 3000,
    ccartPreviewButtonText: configTranslation.preview_design_button_label || 'Xem thiết kế',
    autofixerCallback: () => {
      theme.updateCartSummaries()
    },
    openPreview: () => {
      const previewButton = document.querySelector('.tee-btn--preview');
      if (previewButton) {
        previewButton.click();
      }
    },
    getProcessBar: ({ activeGroupIndex, numberOfGroups }) => {
      const stepsTextConfig = window.teeinblue?.stepsText || 'Step {current_step} of {total_step}';
      const stepsText = stepsTextConfig
        .replaceAll('{current_step}', `${activeGroupIndex + 1}`)
        .replaceAll('{total_step}', `${numberOfGroups}`);

      const steps = Array.from({ length: numberOfGroups }, (_, i) => {
        return `<li class="step ${i === activeGroupIndex ? 'active' : ''}"></li>`;
      }).join('');

      return `
            <span class="step-text">${stepsText}</span>
            <ul class="steps">${steps}</ul>
            <button class="tee-preview" onClick="window.teeinblue.openPreview()"></button>
        `;
    },
    ...(() => {
      const locale = window.Shopify?.locale;
      // default language
      return {
        personalizeButtonText: configTranslation.personalize_button_label || 'Tùy chỉnh áo',
        nextGroupButtonText: 'Next Step',
        backGroupButtonText: 'Back',
        addtocartButtonText: configTranslation.addtocart_button_label || 'ADD TO CART',
        buynowButtonText: configTranslation.buynow_button_label || 'BUY IT NOW',
        subTotalText: 'Tổng thanh toán',
        stepsText: 'Step {current_step} of {total_step}',
        modalTitle: configTranslation.form_header_title || 'PERSONALISE THIS ITEM',
      };
    })(),
  };

  const validate = () => {
    let showMes = false;
    let optionSelectNameError = "";
    
    const optionGroups = document.querySelectorAll('.option-selectors .selector-wrapper');
    
    optionGroups.forEach(group => {
      const labelEl = group.querySelector('label');
      let labelName = "option";
      
      if (labelEl) {
        const clone = labelEl.cloneNode(true);
        const spans = clone.querySelectorAll('span');
        spans.forEach(s => s.remove());
        labelName = clone.textContent.trim().replace(':', '');
      }
      
      let isValid = false;
      
      const customVariant = group.querySelector('.custom-variant');
      if (customVariant) {
        if (customVariant.querySelector('a.actived')) {
          isValid = true;
        }
      } 
      else {
        const fieldset = group.querySelector('fieldset.bss-custom-product-form__input');
        if (fieldset) {
          if (fieldset.querySelector('input[type="radio"]:checked')) {
            isValid = true;
          }
        } else {
          isValid = true; 
        }
      }

      if (!isValid) {
        showMes = true;
        optionSelectNameError = labelName;
      }
    });
    
    if (showMes) {
      const tptMess = document.getElementById("tpt-mess");
      if (tptMess) {
        if (tptMess.classList.contains("show") && typeof mesTimeout !== 'undefined') {
          clearTimeout(mesTimeout);
        }
        
        const messText = (window.variant_select && window.variant_select.error) || "Vui lòng chọn ";
        
        const textEl = document.getElementById("tpt-mess__text");
        if (textEl) {
          textEl.innerHTML = messText + optionSelectNameError.toLowerCase();
        }
        
        tptMess.classList.add("tpt-mess__type-error", "show");
        
        window.mesTimeout = setTimeout(() => {
          tptMess.classList.remove("show");
        }, 4000);
      } else {
        alert("Vui lòng chọn " + optionSelectNameError);
      }
    }
    
    return !showMes;
  }

  document.addEventListener("click", function(e) {
    const personalizeBtn = e.target.closest("button.tee-btn--personalize");

    if (personalizeBtn) {
      if (!validate()) {
        e.preventDefault();
        e.stopPropagation();
      }
    }
  }, true);

  document.addEventListener('DOMContentLoaded', function () {
    document.documentElement.style.setProperty(
      '--personalize-modal-title',
      `"${window.teeinblue?.modalTitle}"` || '"PERSONALISE THIS ITEM"'
    );

    const parsePrice = (text) => {
      if (!text) return 0;

      // Remove everything except digits
      const number = text.replace(/[^\d]/g, '');
      return number ? parseInt(number, 10) : 0;
    };

    const setBuyNowLoading = (btn, isLoading) => {
      if (!btn) return;

      if (isLoading) {
        btn.classList.add('tee-processing');
        btn.disabled = true;

        btn.innerHTML = `
          <span class="tee-spinner tee-spinner--sm">
            <svg xmlns="http://www.w3.org/2000/svg" class="tee-icon-loading tee-icon-loading--sm" viewBox="0 0 128 128">
              <g>
                <path d="M64 127.75A63.76 63.76 0 0 1 52.8 1.23v14.23a49.8 49.8 0 1 0 22.4 0V1.23A63.76 63.76 0 0 1 64 127.75"></path>
                <animateTransform attributeName="transform" dur="1260ms" from="0 64 64" repeatCount="indefinite" to="360 64 64" type="rotate"></animateTransform>
              </g>
            </svg>
          </span>
        `;
      } else {
        btn.classList.remove('tee-processing');
        btn.disabled = false;

        btn.innerHTML = `<span>${configTranslation.buynow_button_label || "BUY IT NOW"}</span>`;
      }
    };

    const injectBuyNowButton = () => {
      const wrapper = document.querySelector('.tee-form-atc-wrapper');
      if (!wrapper) return;

      const atcBtn = wrapper.querySelector('#teeAtcButton');
      if (!atcBtn) return;

      const expectedAtcText = (configTranslation.addtocart_button_label || 'ADD TO CART').trim().toUpperCase();
      
      const isAddToCart = atcBtn.textContent.trim().toUpperCase() === expectedAtcText;

      let buyNowBtn = wrapper.querySelector('.tee-btn--buynow');

      if (isAddToCart) {
        if (!buyNowBtn) {
          buyNowBtn = document.createElement('button');
          buyNowBtn.type = 'button';
          buyNowBtn.className = 'tee-btn tee-btn--buynow';
          buyNowBtn.innerHTML = `<span>${configTranslation.buynow_button_label || "BUY IT NOW"}</span>`;

          buyNowBtn.addEventListener('click', () => {
            window.__teeBuyNow = true;

            setBuyNowLoading(buyNowBtn, true);

            window.teeinblue?.addToCart();
          });

          atcBtn.insertAdjacentElement('afterend', buyNowBtn);
        }

        buyNowBtn.style.display = '';
      } else {
        if (buyNowBtn) {
          buyNowBtn.style.display = 'none';
        }
      }
    };

    const updateTotalPrice = () => {
      const priceSpan = document.querySelector('.product-price .current-price');
      const popupPrice = document.querySelector('.tee-edit-popup--product-price .product-price--value');

      if (!priceSpan) return;

      // Store the original price (only once)
      if (!priceSpan.dataset.originalPrice || priceSpan.dataset.originalPrice === '0') {
        const base = parsePrice(priceSpan.textContent);
        priceSpan.dataset.originalPrice = base;
      }

      const basePrice = parseFloat(priceSpan.dataset.originalPrice);
      const addonsDiv = document.querySelector('.tee-price-addons');

      const addonPrice = (() => {
        if (!addonsDiv) return 0;
        const strong = addonsDiv.querySelector('strong');
        return strong ? parsePrice(strong.textContent) : 0;
      })();

      const total = basePrice + addonPrice;
      const formatted = `${total.toLocaleString('vi-VN')}₫`;

      

      // update popup price
      if (popupPrice && popupPrice.textContent.trim() !== formatted) {
        popupPrice.textContent = formatted;
      }
    };

    const addPriceToPersonalizePoup = (target) => {
      let price = '';

      const priceEl = document.querySelector('.price-item--regular, .current-price');
      if (priceEl) {
        price = priceEl.textContent.trim();
      }

      const subTotalText = window.teeinblue?.subTotalText || 'Subtotal';

      const htmlToInsert = `
      <div class="tee-edit-popup--product-price">
        <span class="product-price--title">${subTotalText}</span>
        <span class="product-price--value">${price}</span>
      </div>
    `;

      target.insertAdjacentHTML('afterbegin', htmlToInsert);
    };

    const scrollToBottom = () => {
      const dialogBody = document.querySelector('.tee-edit-popup > .vm--modal > .tee-dialog-inner > .tee-dialog-body');
      if (dialogBody) {
        dialogBody.scrollTo({
          top: dialogBody.scrollHeight,
          behavior: 'smooth',
        });
      }
    };

    const observer = new MutationObserver((mutations, obs) => {
      const target = document.querySelector('.tee-edit-popup .tee-form-atc-wrapper');
      const subtotal = document.querySelector('.tee-edit-popup--product-price');
      if (target && !subtotal) {
        addPriceToPersonalizePoup(target);
      }

      if (target) {
        injectBuyNowButton();
      }

      if (subtotal) {
        updateTotalPrice();
      }

      for (const mutation of mutations) {
        for (const node of mutation.addedNodes) {
          if (!(node instanceof HTMLElement)) continue;

          if (node.matches && node.matches('.tee-form-responses')) {
            const teeError = node.querySelector('.tee-error');
            if (teeError) {
              window.__teeBuyNow = false;
              const buyNowBtn = document.querySelector('.tee-btn--buynow');
              setBuyNowLoading(buyNowBtn, false);
              
              scrollToBottom();
              return;
            }
          }
        }
      }
    });
    observer.observe(document.body, {
      childList: true,
      subtree: true,
    });

    const debounce = (func, delay) => {
      let timeout;
      return function (...args) {
        clearTimeout(timeout);
        timeout = setTimeout(() => func.apply(this, args), delay);
      };
    };

    const handleInput = (e) => {
      const grandparent = e.target.parentElement?.parentElement;
      const grandparentClasses = Array.from(grandparent?.classList || []).filter((cls) => cls !== 'tee-field');
      const matchingElements = Array.from(document.querySelectorAll('.tee-field'))
        .filter((el) => el !== grandparent && grandparentClasses.some((cls) => el.classList.contains(cls)))
        .filter((el) => el.querySelector('select.tee-field__select'));

      const addonChargeData = {};
      matchingElements.forEach((el, elIndex) => {
        const select = el.querySelector('select.tee-field__select');
        if (select) {
          const layerId = select.id;
          const options = select.options;
          let value = null;
          for (let i = 0; i < options.length; i++) {
            if (options[i].text?.toLowerCase() === e.target.value[elIndex]?.toLowerCase()) {
              value = options[i].value;
              break;
            }
          }
          addonChargeData[layerId] = value;
        }
      });

      const pId = window.teeinblueCampaign?.productId;
      if (pId) {
        let customizations = localStorage.getItem('teeinblue-customizations');
        customizations = customizations ? JSON.parse(customizations) : [];
        const cIndex = customizations.findIndex((c) => c.productId == pId);
        if (cIndex > -1) {
          customizations[cIndex].customization = {
            ...customizations[cIndex].customization,
            ...addonChargeData,
            [e.target.id]: e.target.value,
          };
          localStorage.setItem('teeinblue-customizations', JSON.stringify(customizations));
          window.teeinblue.refillCustomization(pId);
        }
      }
    };

    const debouncedHandleInput = debounce((e) => handleInput(e), 500);

    // Clean up after paste or input
    document.addEventListener('input', function (e) {
      if (e.target.matches('.letters-only .tee-field__input > input')) {
        e.target.value = e.target.value.replace(/[^a-zA-Z\-\. ]/g, '');
      }

      if (e.target.matches('.tee-field__input > input')) {
        debouncedHandleInput(e);
      }
    });

    // Block invalid keys
    document.addEventListener('keydown', function (e) {
      if (e.target.matches('.letters-only .tee-field__input > input')) {
        const allowedKeys = /^[a-zA-Z\-\. ]$/;
        if (!allowedKeys.test(e.key) && e.key.length === 1) {
          e.preventDefault();
        }
      }
    });

    // Prevent paste if it contains invalid characters
    document.addEventListener('paste', function (e) {
      if (e.target.matches('.letters-only .tee-field__input > input')) {
        const pasted = e.clipboardData.getData('text');
        if (/[^a-zA-Z\-\. ]/.test(pasted)) {
          e.preventDefault();
        }
      }
    });

    // Prevent drag-and-drop into input
    document.addEventListener('drop', function (e) {
      if (e.target.matches('.letters-only .tee-field__input > input')) {
        e.preventDefault();
      }
    });

    document.addEventListener('teeinblue-event-after-cart-added', () => {
      const buyNowBtn = document.querySelector('.tee-btn--buynow');
      window.teeinblue?.removeCustomization?.(window.teeinblueCampaign?.productId);
      theme.updateCartSummaries()

      if (window.__teeBuyNow) {
        setBuyNowLoading(buyNowBtn, false);

        if (document.querySelector('.tee-error')) {
          window.__teeBuyNow = false;
          return;
        }

        setTimeout(() => {
          window.__teeBuyNow = false;
          window.location.href = '/checkout';
        }, 3000)
        
      } else {
        document.querySelector('.toggle-cart-summary')?.click();
        
        setTimeout(() => {
          document.querySelector('.tee-edit-popup .tee-dialog-close')?.click()
        }, 1000)
        
      }
    });

    window.addEventListener(
      'keyup',
      function (e) {
        if (event.which === 27) {
          // 27 is Escape key
          const modals = document.querySelectorAll('.vm--modal');
          if (modals.length > 0) {
            e.stopImmediatePropagation();

            const topModal = modals[modals.length - 1];
            const closeBtn = topModal.querySelector('.tee-dialog-close');
            if (closeBtn) {
              closeBtn.click();
            }
          }
        }
      },
      true
    );

    window.addEventListener(
      'keydown',
      function (e) {
        if (e.which === 39) {
          // 39 is arrow right key
          const teeNextSlideButton = document.querySelector('.tee-slider__button--next');
          if (teeNextSlideButton) {
            e.stopImmediatePropagation();
            teeNextSlideButton.click();
          }
        }

        if (e.which === 37) {
          // 37 is arrow left key
          const teePrevSlideButton = document.querySelector('.tee-slider__button--prev');
          if (teePrevSlideButton) {
            e.stopImmediatePropagation();
            teePrevSlideButton.click();
          }
        }
      },
      true
    );
  });
</script>
